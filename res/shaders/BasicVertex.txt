#version 330 core  

layout (location = 0) in vec3 position;
layout (location = 1) in vec4 inColor;
layout (location = 2) in vec3 normal;

out vec4 ourColor;
out vec3 Normal;

uniform mat4 u_MVP;
uniform vec4 u_Tint;

void main()
{
    gl_Position = u_MVP * vec4(position.x, position.y, position.z, 1.0);
    ourColor = u_Tint * inColor;
    Normal = normal;
};
