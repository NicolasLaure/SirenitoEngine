#version 330 core  
out vec4 FragColor;

in vec4 ourColor;
in vec2 ourTextureCoordinates;
in vec3 Normal;
in vec3 FragPos;


uniform sampler2D u_Texture;

uniform vec3 u_AmbientLightColor;

uniform vec3 u_PointLightColor;
uniform vec3 u_PointLightPos;

void main()
{
	vec4 tintedColor = texture(u_Texture,  ourTextureCoordinates) * ourColor;
	
	vec3 norm = normalize(Normal);
	vec3 lightDir = normalize(u_PointLightPos - FragPos);
	
	float diff = max(dot(norm, lightDir), 0.0);
	vec3 diffuse = diff * u_PointLightColor;

	vec3 result = vec3(vec4((u_AmbientLightColor + diffuse),1.0) * tintedColor); 
	
	FragColor = vec4(result, tintedColor.a);
};
